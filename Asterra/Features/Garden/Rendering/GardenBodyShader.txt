// Garden bodies: zoom-aware instanced quads for the collected objects (Night Sea pass 1).
// Appended to NightSeaShader.txt by NightSeaRenderer only (the wormhole keeps gardenPoint*).
//
// At the default field of view every object looks exactly like the kit sprites (pinpoint stars
// with glow and spikes, small planet glows). Zooming in grows the apparent disk and blends to:
//   - stars and the Sun: a live photosphere (StarKit look: granulation, limb darkening, corona)
//   - Moon and planets: their equirectangular texture on a sphere lit by the real Sun
//     (phase angle from Astronomy Engine, bright limb toward the Sun), Saturn with bands + rings.
// Sizes must match NightSeaObject.apparent(...) in Swift, which places labels and tap targets.

struct GardenBody {
    float4 dirSize;   // xyz direction, w kit radius (pt)
    float4 color;     // rgb, w kind: 0 star, 1 planet, 2 moon, 3 sun, + 10 when selected
    float4 extra;     // x phase angle (rad), y texture slice (-1 none), z spin (rad/s), w axial tilt (rad)
    float4 sun;       // xyz Sun direction
};

struct BodyOut {
    float4 position [[position]];
    float4 color;
    float4 extra;
    float4 geom;      // x disk / core radius (pt), y quad radius (pt), z detail 0…1, w kit sprite radius (pt)
    float2 v;         // quad coordinates −1…1, y down
    float2 limb;      // bright-limb direction on screen (y down)
    float kind;
};

// Zoom relative to the kit's 72° view. Shared by both size rules.
static float gbZoom(constant SeaUniforms &u) {
    float minSide = min(u.res.x, u.res.y) / u.res.z;
    float fov = 2.0 * atan(0.62 * minSide / u.cam.x) * 180.0 / PI;
    return 72.0 / max(fov, 0.5);
}

// Growth exponent: √zoom like the kit up to ~2.5×, then the disk scales with the optics.
static float gbGrowth(float Z, float lo) { return mix(lo, 1.0, smoothstep(2.5, 12.0, Z)); }

vertex BodyOut gardenBodyVertex(uint vid [[vertex_id]], uint iid [[instance_id]],
                                constant SeaUniforms &u [[buffer(0)]],
                                const device GardenBody *bodies [[buffer(1)]]) {
    GardenBody g = bodies[iid];
    float3 d = g.dirSize.xyz;
    BodyOut o;
    o.position = float4(2, 2, 0, 1); o.color = float4(0); o.extra = float4(0); o.geom = float4(0);
    o.v = float2(0); o.limb = float2(0); o.kind = g.color.w;
    if (d.z < -0.01) return o;
    float3 projected = projectDir(d, u, 0.06);          // pinhole or the stereographic dome
    if (projected.z < 0.5) return o;
    float2 q = projected.xy;
    float kind = fmod(g.color.w, 10.0);
    float Z = gbZoom(u);
    float disk, quad, detail, kit;
    if (kind < 0.5) {
        // A star stays a point of light when zoomed: the kit's √zoom growth, capped at 2.2×
        // (GardenZoomLimits.maxStarGrowth). Only the Sun, Moon and planets resolve into disks.
        float core = g.dirSize.w * min(sqrt(max(Z, 0.0)), 2.2);
        detail = 0.0;
        kit = max(core * 8.0, 5.0 + core + 2.0);
        disk = core;
        quad = mix(kit, core * 2.6, detail);
        // Naked-eye look needs the altitude (scintillation) and a steady per-star seed.
        g.extra = float4(d.z, fract(sin(dot(d, float3(12.9898, 78.233, 37.719))) * 43758.5453) * 100.0, 0, 0);
    } else {
        float spriteR = g.dirSize.w;
        float frac = kind > 1.5 ? 0.24 : 0.13;
        float d0 = spriteR * frac;
        if (kind > 1.5) {
            // Moon and Sun: the kit's enlarged disk, never smaller than their true 0.27° radius,
            // so a deep zoom frames them like binoculars instead of overflowing the screen.
            disk = max(d0 * pow(max(Z, 1.0), 0.35), u.cam.x * 0.004652);
            // The Moon resolves early; the Sun stays a blinding glare until zoomed in
            // (like looking through a solar filter).
            detail = kind > 2.5 ? smoothstep(14.0, 40.0, disk) : smoothstep(3.0, 10.0, disk);
        } else {
            // Planets are far below a degree: enlarged so the zoom can reveal them.
            disk = d0 * pow(max(Z, 1.0), gbGrowth(Z, 0.35));
            detail = smoothstep(4.0, 14.0, disk);
        }
        kit = kind > 2.5 ? disk * 9.0 : disk / frac;   // the Sun's glare reaches far
        // Saturn's rings and the Sun's corona need room beyond the disk.
        float body = disk * (kind > 2.5 ? 2.6 : g.extra.y < -0.5 ? 2.5 : 1.7);
        quad = mix(kit, body, detail);
    }
    // Showcase cap: a resolved body never exceeds ~30% of the short side, so it stays whole
    // (with its corona or rings) and the per-pixel photosphere stays cheap.
    float cap = 0.3 * min(u.res.x, u.res.y) / u.res.z;
    if (disk > cap) { quad *= cap / disk; disk = cap; }
    float2 corner = float2((vid & 1) ? 1.0 : -1.0, (vid & 2) ? 1.0 : -1.0);
    float2 px = (q + corner * quad) * u.res.z;
    o.position = float4(px.x / u.res.x * 2 - 1, 1 - px.y / u.res.y * 2, 0, 1);
    o.v = corner;
    o.color = float4(g.color.rgb, 1);
    o.extra = g.extra;
    o.geom = float4(disk, quad, detail, kit);
    // Bright limb: the great-circle direction from the object toward the Sun, on screen.
    float3 s = g.sun.xyz;
    float3 perp = s - dot(s, d) * d;
    float2 b = float2(dot(perp, u.R.xyz), -dot(perp, u.U.xyz));
    o.limb = length(b) > 1e-5 ? normalize(b) : float2(-0.7, -0.7);
    return o;
}

// ---------- the kit look for planets (identical to gardenPointFragment) ----------

static float4 gbKitBody(float r, float kind, bool selected, float3 col, float time) {
    if (r > 1.0) return float4(0);
    bool disc = kind > 1.5;
    float coreR = disc ? 0.24 : 0.13;
    float coreA = 1 - smoothstep(coreR * 0.75, coreR, r);
    float glow = exp(-r * r * (disc ? 9.0 : 16.0)) * 0.55 + exp(-r * r * 60.0) * 0.5;
    float ring = selected ? smoothstep(0.03, 0.0, abs(r - 0.92)) * 0.9 : 0.0;
    float breathe = disc ? 1.0 : 0.92 + 0.08 * sin(time * 1.1 + col.g * 23.0);
    glow *= breathe;
    if (!disc && !selected) ring += smoothstep(0.035, 0.0, abs(r - 0.62)) * 0.16;
    float3 coreColor = mix(col, float3(1), disc ? 0.35 : 0.6);
    float3 c = col * glow + coreColor * coreA + float3(ring);
    float a = clamp(coreA + glow + ring, 0.0, 1.0);
    return float4(min(c, float3(a)), a);
}

// ---------- naked-eye look when zoomed out ----------
// Light is returned with alpha only on the opaque core, so the glow ADDS to the sky behind it
// (premultiplied over), the way starlight and glare blend into a real night sky.

// Atmospheric scintillation: fast brightness flicker, strongest near the horizon.
static float gbScintillation(float altitude) { return mix(0.45, 0.10, smoothstep(0.05, 0.8, altitude)); }

static float gbTwinkle(float t, float seed, float altitude) {
    float n = 0.5 * sin(t * 7.3 + seed) + 0.3 * sin(t * 13.1 + seed * 1.7) + 0.2 * sin(t * 23.7 + seed * 2.3);
    return 1.0 + gbScintillation(altitude) * n;
}

// A star as the eye sees it: a tiny white-hot point (Gaussian) in a Moffat seeing halo
// tinted by its colour (with a slight colour flicker), plus a faint wide glow for bright stars.
// No diffraction spikes: those belong to cameras and telescopes, not eyes.
static float4 gbEyeStar(float2 vp, float core, float3 col, float t, float altitude, float seed) {
    float r = length(vp);
    float b = saturate((core - 1.0) / 4.0);                    // brightness from the kit radius
    float sigma = 0.5 + 0.22 * core;
    float tw = gbTwinkle(t, seed, altitude);
    float amp = gbScintillation(altitude);
    float3 tint = col * (1.0 + amp * 0.35 * float3(sin(t * 9.1 + seed), sin(t * 11.3 + seed * 1.3), sin(t * 8.7 + seed * 0.7)));
    float point = exp(-(r * r) / (sigma * sigma));
    float halo = (0.25 + 0.55 * b) * pow(1.0 + (r * r) / (sigma * sigma * 6.25), -2.2);
    float wide = 0.10 * b * exp(-r / (sigma * 9.0));
    float3 c = mix(tint, float3(1.0), 0.7) * point * mix(1.0, tw, 0.6) + tint * (halo + wide) * tw;
    return float4(c, point);
}

// The Sun zoomed out: an overexposed white disk with a warm, far-reaching glare and soft rays.
static float4 gbSunGlare(float2 vp, float disk, float t, float edgeFade) {
    float r = length(vp) / disk;
    float ang = atan2(vp.y, vp.x);
    float core = 1.0 - smoothstep(0.9, 1.05, r);
    float glow = 0.85 / (1.0 + pow(max(r - 0.9, 0.0) * 1.6, 2.0)) + 0.35 * exp(-r * 0.35);
    // Soft glare streaks (eye/lens scatter), not crisp spikes.
    float rays = pow(0.5 + 0.5 * sin(ang * 13.0 + 2.0 * sin(ang * 5.0 + t * 0.3)), 3.0) * 0.10 * exp(-r * 0.5);
    float3 glowCol = mix(float3(1.0, 0.86, 0.60), float3(1.0, 0.96, 0.86), exp(-max(r - 1.0, 0.0)));
    float3 c = float3(1.0, 0.98, 0.94) * core + glowCol * (glow + rays) * (1.0 - core) * edgeFade;
    return float4(c, core);
}

// ---------- photosphere (StarKit's shader, in units of the disk radius) ----------

static float gbHash3(float3 p) { p = fract(p * 0.3183099 + 0.1); p *= 17.0; return fract(p.x * p.y * p.z * (p.x + p.y + p.z)); }

static float gbNoise3(float3 x) {
    float3 i = floor(x); float3 f = fract(x); f = f * f * f * (f * (f * 6.0 - 15.0) + 10.0);
    return mix(mix(mix(gbHash3(i), gbHash3(i + float3(1,0,0)), f.x),
                   mix(gbHash3(i + float3(0,1,0)), gbHash3(i + float3(1,1,0)), f.x), f.y),
               mix(mix(gbHash3(i + float3(0,0,1)), gbHash3(i + float3(1,0,1)), f.x),
                   mix(gbHash3(i + float3(0,1,1)), gbHash3(i + float3(1,1,1)), f.x), f.y), f.z);
}

static float gbFbm(float3 p) {
    float a = 0.5; float s = 0.0;
    for (int i = 0; i < 5; i++) { s += a * gbNoise3(p); p = p * 2.03 + float3(1.7, 9.2, 3.1); a *= 0.5; }
    return s;
}

static float2 gbWorley(float3 p, float t) {
    float3 i = floor(p); float3 f = fract(p);
    float d1 = 8.0; float d2 = 8.0;
    for (int x = -1; x <= 1; x++)
    for (int y = -1; y <= 1; y++)
    for (int z = -1; z <= 1; z++) {
        float3 b = float3(float(x), float(y), float(z));
        float3 h = float3(gbHash3(i + b), gbHash3(i + b + float3(17.1, 3.3, 9.7)), gbHash3(i + b + float3(5.7, 23.1, 1.9)));
        float3 o = 0.5 + 0.4 * sin(t * 0.35 + 6.2831 * h);
        float3 rr = b + o - f;
        float dd = dot(rr, rr);
        if (dd < d1) { d2 = d1; d1 = dd; } else if (dd < d2) { d2 = dd; }
    }
    return float2(sqrt(d1), sqrt(d2));
}

static float3 gbRotX(float3 v, float a) { float c = cos(a); float s = sin(a); return float3(v.x, c*v.y - s*v.z, s*v.y + c*v.z); }
static float3 gbRotY(float3 v, float a) { float c = cos(a); float s = sin(a); return float3(c*v.x + s*v.z, v.y, -s*v.x + c*v.z); }

// p in disk radii (y up), px = one screen pixel in disk radii, fade = quad edge fade.
// Same parameters as Asterra's StarTuning.asterra: granulation 0.3, turbulence 0.38, pulse ×2.2.
static float4 gbPhotosphere(float2 p, float px, float t, float3 col, bool sun, float fade) {
    float3 deep = pow(col, float3(2.5)) * 0.75;
    float3 core = col + (1.0 - col) * 0.35;
    float3 glowCol = pow(col, float3(1.6));
    float hot = saturate((col.b - col.r) * 1.4 + 0.55);
    if (sun) {
        float3 sunlight = float3(1.0, 0.93, 0.70);
        core = mix(core, sunlight, 0.7); glowCol = mix(glowCol, sunlight, 0.5);
        deep = mix(deep, sunlight * float3(0.95, 0.72, 0.42), 0.85); hot = 0.3;
    }
    float granAmt = 0.3, turb = 0.45, glowAmt = sun ? 1.0 : 0.85, pulseAmt = 1.32;
    float spotAmt = 0.45 * (1.0 - hot) * (sun ? 0.25 : 1.0);

    float beat = sin(t * 1.2566);
    float pulse = 1.0 + pulseAmt * 0.12 * beat;
    float r = length(p);
    float ang = atan2(p.y, p.x);
    float d = max(r - 1.0, 0.0);

    float a1 = gbFbm(float3(cos(ang) * 1.6, sin(ang) * 1.6, t * 0.05));
    float a2 = gbFbm(float3(cos(ang) * 4.5, sin(ang) * 4.5, d * 2.5 - t * 0.18));
    float streak = a1 * a1 * 1.8 + a2 * 0.55;
    float corona = exp(-d * 3.2 / glowAmt) * (0.3 + streak) * pulse;
    float inner = exp(-d * 16.0);
    float bloom = 0.22 / (1.0 + pow(d * 5.0 / glowAmt, 2.0)) * pulse;
    float3 outc = glowCol * (corona * 0.95 * glowAmt + inner * 1.3 + bloom * 0.7 * glowAmt) * fade;

    float disk = 0.0;
    if (r < 1.0) {
        float zz = sqrt(1.0 - r * r);
        float3 q = gbRotY(float3(p, zz), t * 0.06);
        float warp = gbFbm(q * 1.6 + float3(0.0, 0.0, t * 0.04));
        float3 qw = q + (float3(warp, gbFbm(q * 1.6 + 4.3), gbFbm(q * 1.6 + 8.1)) - 0.5) * 0.35;
        float sc = 12.0 * pow(turb, 1.5);
        float lane = 1.0 + 1.2 * (1.0 - granAmt);
        float2 w1 = gbWorley(qw * sc, t);
        float2 w2 = gbWorley(qw * sc * 0.38 + 3.1, t * 0.6);
        float gran = smoothstep(0.0, 0.5 * lane, w1.y - w1.x) * (0.7 + 0.3 * (1.0 - w1.x));
        float cell = smoothstep(0.0, 0.6 * lane, w2.y - w2.x) * (0.75 + 0.25 * (1.0 - w2.x));
        float big = gbFbm(qw * 2.2 + float3(t * 0.05, -t * 0.03, t * 0.02));
        float gc = mix(0.42, 0.1, hot) * granAmt;
        float I = 0.66 + (big - 0.5) * 0.6 + (gran - 0.5) * gc + (cell - 0.5) * gc * 0.6;
        float spot = smoothstep(0.63, 0.72, gbFbm(q * 3.0 + float3(11.0, 3.0, t * 0.01)));
        I *= 1.0 - spot * spotAmt;
        float fac = smoothstep(0.55, 0.75, gbFbm(q * 5.0 + float3(5.0))) * pow(1.0 - zz, 1.5);
        I += fac * 0.3;
        float m = 1.0 - zz;
        float limb = 1.0 - 0.5 * m - 0.2 * m * m;
        float3 surf = mix(deep, core, smoothstep(0.3, 1.0, I)) * I * 2.4;
        surf = mix(surf, deep * I * 2.0, m * 0.55);
        surf *= limb;
        surf += core * smoothstep(0.85, 1.1, I) * 0.8 * granAmt;
        surf += glowCol * smoothstep(0.9, 1.0, r) * 0.8 * mix(0.35, 1.0, granAmt);
        surf *= 1.0 + pulseAmt * 0.05 * beat;
        disk = 1.0 - smoothstep(1.0 - 1.5 * px, 1.0, r);
        outc = mix(outc, surf, disk);
    }
    float L = dot(outc, float3(0.2126, 0.7152, 0.0722));
    float Lt = 1.0 - exp(-L * 1.5);
    outc = outc * (Lt / max(L, 1e-4));
    outc = mix(outc, float3(1.0), smoothstep(0.8, 1.0, Lt) * 0.55);
    outc = clamp(outc, 0.0, 1.0);
    float alpha = max(disk, max(outc.x, max(outc.y, outc.z)));
    return float4(outc, alpha);
}

// ---------- planets and the Moon (Planet.metal, lit by the real Sun) ----------

static float gbHash2(float2 p) { return fract(sin(dot(p, float2(127.1, 311.7))) * 43758.5453); }
static float gbNoise2(float2 x) {
    float2 i = floor(x); float2 f = fract(x); f = f * f * (3.0 - 2.0 * f);
    return mix(mix(gbHash2(i), gbHash2(i + float2(1, 0)), f.x),
               mix(gbHash2(i + float2(0, 1)), gbHash2(i + float2(1, 1)), f.x), f.y);
}

static float3 gbBands(float lat, float lon) {
    float3 tint = float3(0.86, 0.76, 0.56);
    float wobble = 0.08 * sin(lon * 3.0 + lat * 9.0) + 0.06 * (gbNoise2(float2(lon * 4.0, lat * 30.0)) - 0.5);
    float b = 0.5 + 0.5 * sin((lat + wobble) * 22.0);
    float fine = gbNoise2(float2(lon * 2.0, lat * 80.0));
    float3 c = mix(tint * 1.12, tint * float3(0.82, 0.72, 0.55), b * 0.55 + fine * 0.12);
    return mix(c, tint * float3(0.70, 0.74, 0.80), smoothstep(1.15, 1.5, abs(lat)));
}

static float4 gbRing(float2 q, float px) {
    float rho = length(float2(q.x, q.y / 0.30));
    if (rho < 1.28 || rho > 2.25) return float4(0.0);
    float edge = smoothstep(1.28, 1.34, rho) * (1.0 - smoothstep(2.18, 2.25, rho));
    float cassini = 1.0 - 0.85 * smoothstep(1.86, 1.89, rho) * (1.0 - smoothstep(1.95, 1.98, rho));
    float crepe = mix(0.35, 1.0, smoothstep(1.28, 1.55, rho));
    float grooves = 0.82 + 0.18 * (gbNoise2(float2(rho * 38.0, 0.5)) - 0.5) * 2.0 + 0.06 * sin(rho * 23.0);
    float a = edge * cassini * crepe * grooves * 0.9;
    float3 c = float3(0.86, 0.76, 0.56) * float3(1.05, 0.98, 0.86) * (0.75 + 0.25 * smoothstep(1.4, 2.1, rho));
    return float4(c * a, a);
}

static float4 gbOver(float4 top, float4 bottom) { return top + bottom * (1.0 - top.a); }

// Thick atmospheres scatter light a little past the terminator.
static float rimGlow(float z, float atmoAmt, float ndl) {
    return atmoAmt * 0.6 * pow(1.0 - z, 2.0) * smoothstep(-0.35, 0.0, ndl);
}

// p in disk radii (y up). limb: bright-limb direction (y up). phase: Sun–body–Earth angle.
static float4 gbPlanet(float2 p, float px, float t, float4 extra, float2 limb,
                       texture2d_array<float> tex, float lod, float fade) {
    constexpr sampler s(address::repeat, filter::linear, mip_filter::linear);
    float slice = extra.y;
    float phase = extra.x;
    // atmosphere per body: Moon 0, Mercury 1, Venus 2, Mars 3, Jupiter 4, Saturn −1
    float3 atmo = float3(0); float atmoAmt = 0.0;
    if (slice > 1.5 && slice < 2.5) { atmo = float3(1.0, 0.86, 0.60); atmoAmt = 0.75; }
    else if (slice > 2.5 && slice < 3.5) { atmo = float3(1.0, 0.62, 0.45); atmoAmt = 0.30; }
    else if (slice > 3.5) { atmo = float3(0.95, 0.85, 0.72); atmoAmt = 0.45; }
    else if (slice < -0.5) { atmo = float3(0.98, 0.90, 0.72); atmoAmt = 0.40; }
    bool rings = slice < -0.5;

    float3 L = normalize(float3(sin(phase) * limb, cos(phase)));
    float ca = cos(extra.w), sa = sin(extra.w);
    float2 q = float2(ca * p.x + sa * p.y, -sa * p.x + ca * p.y);
    float r = length(p);

    float4 ringCol = rings ? gbRing(q, px) : float4(0.0);
    ringCol.rgb *= 0.35 + 0.65 * smoothstep(-0.3, 0.6, dot(normalize(float3(q, 0.3)), L));
    bool ringFront = q.y < 0.0;

    float d = max(r - 1.0, 0.0);
    float litSide = smoothstep(-0.4, 0.8, dot(normalize(float3(p, 0.0001)), L));
    // Only outside the limb (overlapping its anti-aliased edge): behind a transparent night
    // side the angle-based haze would otherwise show as rays from the centre.
    float halo = exp(-d * 18.0) * atmoAmt * 0.55 * litSide * fade * smoothstep(1.0 - 2.0 * px, 1.0, r);
    float4 outc = float4(atmo * halo, halo);
    if (!ringFront) outc = gbOver(outc, ringCol);

    if (r < 1.0 + 2.0 * px) {
        float rr = min(r, 1.0);
        float z = sqrt(max(1.0 - rr * rr, 0.0));
        float3 n = float3(p, z);
        float3 nt = float3(ca * n.x + sa * n.y, -sa * n.x + ca * n.y, n.z);
        float lon = atan2(nt.x, nt.z) + t * extra.z;
        float lat = asin(clamp(nt.y, -1.0, 1.0));
        // Longitude texels crowd toward the poles of an equirectangular map: blur them there.
        float poleLod = lod + log2(1.0 / max(cos(lat), 0.05));
        float3 albedo = slice > -0.5
            ? tex.sample(s, float2(fract(lon / 6.2831853 + 0.5), 0.5 - lat / 3.1415927), uint(slice + 0.5), level(poleLod)).rgb
            : gbBands(lat, lon);
        float ndl = dot(n, L);
        // airless bodies: crisp terminator; thick atmospheres: softer and limb-darkened
        float soft = mix(0.04, 0.25, saturate(atmoAmt * 1.4));
        float terminator = smoothstep(-soft, soft * 1.5, ndl);
        float diffuse = mix(pow(max(ndl, 0.0), 0.7), 1.0, 0.12) * terminator;
        float limbDark = mix(1.0, 0.55 + 0.45 * z, saturate(atmoAmt * 1.4));
        bool moon = slice > -0.5 && slice < 0.5;
        float earthshine = moon ? 0.06 : 0.0;
        float3 surf = albedo * (earthshine + 1.05 * diffuse * limbDark);
        surf += atmo * pow(1.0 - z, 3.0) * atmoAmt * smoothstep(-0.2, 0.5, ndl) * 0.9;
        float disk = 1.0 - smoothstep(1.0 - 1.5 * px, 1.0, r);
        // The night side is not a black hole in the sky: planets fade into it, the Moon's
        // earthshine-lit side only dims the sky behind it.
        float lit = saturate(terminator + rimGlow(n.z, atmoAmt, ndl));
        float opacity = disk * mix(moon ? 0.55 : 0.0, 1.0, lit);
        outc = gbOver(float4(clamp(surf, 0.0, 1.0) * opacity, opacity), outc);
    }
    if (ringFront) outc = gbOver(ringCol, outc);
    return clamp(outc, 0.0, 1.0);
}

fragment float4 gardenBodyFragment(BodyOut in [[stage_in]], constant SeaUniforms &u [[buffer(0)]],
                                   texture2d_array<float> tex [[texture(0)]]) {
    float2 v = in.v;
    float rq = length(v);
    if (rq > 1.0) discard_fragment();
    float kind = fmod(in.kind, 10.0);
    bool selected = in.kind >= 10.0;
    float3 col = in.color.rgb;
    float disk = in.geom.x, quad = in.geom.y, detail = in.geom.z, kit = in.geom.w;
    float time = u.res.w;
    float2 vp = v * quad;                              // points from the centre, y down
    float fade = 1.0 - smoothstep(0.8, 1.0, rq);
    float px = 1.0 / max(disk * u.res.z, 1.0);         // one pixel in disk radii

    float4 o;
    if (kind < 0.5) {
        o = detail < 0.999 ? gbEyeStar(vp, disk, col, time, in.extra.x, in.extra.y) * fade : float4(0);
        if (detail > 0.001) {
            float2 p = float2(vp.x, -vp.y) / disk;
            float4 star = gbPhotosphere(p, px, time * 2.27, col, false, fade);
            o = mix(o, star, detail);
        }
    } else {
        o = detail >= 0.999 ? float4(0)
            : kind > 2.5 ? gbSunGlare(vp, disk, time, fade)
            : gbKitBody(length(vp) / kit, kind, false, col, time);
        if (detail > 0.001) {
            float2 p = float2(vp.x, -vp.y) / disk;
            float4 body;
            if (kind > 2.5) {
                body = gbPhotosphere(p, px, time * 2.27, col, true, fade);
            } else {
                float lod = max(0.0, log2(256.0 / max(2.0 * disk * u.res.z, 1.0)));
                body = gbPlanet(p, px, time, in.extra, float2(in.limb.x, -in.limb.y), tex, lod, fade);
            }
            o = mix(o, body, detail);
        }
    }
    if (selected) {
        // Selection ring just outside the body (or at the kit radius while it is still a point).
        float ringR = mix(kind > 2.5 ? disk * 1.6 : kit * 0.92, disk * 1.3, detail);
        float ring = smoothstep(1.2, 0.0, abs(length(vp) - ringR)) * 0.9;
        o = gbOver(float4(float3(ring), ring), o);
    }
    // Glow may carry more light than alpha (it adds to the sky); keep values in range only.
    return float4(clamp(o.rgb, 0.0, 1.0), clamp(o.a, 0.0, 1.0));
}
