// Wormhole handover (camera -> Night Sea). Compiled together with NightSeaShader.txt, appended after it,
// so it reuses SeaUniforms, skyCol, island, road, sparkle, toScreen and h2 exactly as the garden renders them.
// Units: points (px = fragment / res.z). Centre C, radius R; u = d/R.
//  - outside (u >= 1): the frozen camera frame, gravitationally lensed, with a travelling ripple, twist,
//    radial blur, chromatic taps, a dark contact meniscus at the rim and grain;
//  - inside (u < 1): the Night Sea seen BY DIRECTION, bent around the ray through C.

struct WormUniforms {
    float4 wh;    // C.x, C.y (pt), R (pt), on
    float4 ew;    // E, W, swirl, blur
    float4 wave;  // ripple radius, amplitude, width; glow
    float4 fx;    // grain, new star (0..1), exposure, -
    float4 camT;  // frozen image shift x, y (pt), scale, -
    float4 scr;   // screen W, H (pt), image centre x, y (pt)
};

float2 wormCamUV(float2 p, constant WormUniforms &w) {
    float2 q = w.scr.zw + (p - w.scr.zw) * w.camT.z + w.camT.xy;
    return clamp(q / w.scr.xy, 0.0, 1.0);
}

// Same colour as the garden for direction d (shared envColor), with the object layers looked up where d projects
// (the collection, plus the new star fading in).
float3 wormEnv(float3 d, constant SeaUniforms &u, texture2d<float> tA, texture2d<float> tB, float starAmt, texture2d<float> tMW,
               texture2d<float> tRip) {
    return envColor(d, toScreen(d, u), u, tA, tB, starAmt, tMW, tRip);
}

fragment float4 wormholeFragment(VOut in [[stage_in]], constant SeaUniforms &u [[buffer(0)]], constant WormUniforms &w [[buffer(1)]],
                                 texture2d<float> tA [[texture(0)]], texture2d<float> tMW [[texture(1)]],
                                 texture2d<float> tCam [[texture(2)]], texture2d<float> tB [[texture(3)]],
                                 texture2d<float> tRip [[texture(4)]]) {
    constexpr sampler cs(filter::linear, address::clamp_to_edge);
    float2 fc = in.position.xy;
    float2 px = fc / u.res.z;
    float foc = u.cam.x, uCx = u.cam.y, uCy = u.cam.z;
    float3 F = u.F.xyz, Rv = u.R.xyz, Uv = u.U.xyz;
    float3 d = normalize(F + ((px.x - uCx) / foc) * Rv + ((uCy - px.y) / foc) * Uv);
    float T = u.res.w;
    float noise = (h2(fc + fract(T * 7.0) * 91.0) - .5) * .045;
    float2 C = w.wh.xy;
    float Rw = max(w.wh.z, 1e-3);
    float2 q = px - C;
    float dd = length(q);
    float wu = dd / Rw;
    float ang = atan2(q.y, q.x);
    float swirl = w.ew.z, W = w.ew.y;

    if (wu >= 1.0) {
        float E = w.ew.x * (1.0 + .012 * sin(ang * 3.0 + T * 1.1) * exp(-(wu - 1.0) * 1.2));
        float ds = dd - E * E * dd / (dd * dd + 30.0);
        float xw = (dd - w.wave.x) / max(w.wave.z, 1.0);
        ds += w.wave.y * sin(xw * 2.6) * exp(-xw * xw);
        float tw = swirl * E * E / (dd * dd + E * E * .35 + 1.0);
        float3 cc = float3(0);
        for (int i = 0; i < 3; i++) {
            float fi = float(i);
            float dsi = ds - w.ew.w * fi * .5;
            float ai = ang + tw + (fi - 1.0) * (.006 + .05 * exp(-(wu - 1.0) * 6.0)) * min(1.0, E / max(dd, 1.0));
            for (int k = 0; k < 3; k++) {
                float dk = dsi * (1.0 + .006 * (float(k) - 1.0) * min(1.0, E / max(dd, 1.0) * 2.0));
                float3 c3 = tCam.sample(cs, wormCamUV(C + float2(cos(ai), sin(ai)) * dk, w)).rgb;
                if (k == 0) cc.r += c3.r; else if (k == 1) cc.g += c3.g; else cc.b += c3.b;
            }
        }
        cc /= 3.0;
        float ed = dd - Rw;
        cc *= mix(.25, 1.0, smoothstep(0.0, 3.5, ed));
        cc += float3(.20, .28, .62) * .16 * exp(-max(wu - 1.0, 0.0) * 4.0) * w.wave.w;
        cc += noise * w.fx.x;
        return float4(clamp(cc, 0.0, 1.0), 1.0);
    }

    float3 aF = normalize(F + ((C.x - uCx) / foc) * Rv + ((uCy - C.y) / foc) * Uv);
    float ap = acos(clamp(dot(d, aF), -1.0, 1.0));
    float3 pp = d - aF * dot(d, aF);
    pp = length(pp) > 1e-5 ? normalize(pp) : Rv;
    float3 p2 = cross(aF, pp);
    float tw = swirl * .15 * (1.0 - wu * wu) * W;
    pp = cos(tw) * pp + sin(tw) * p2;
    float aw = 2.75 * pow(wu, 1.18), al = mix(ap, aw, W);
    d = normalize(cos(al) * aF + sin(al) * pp);

    float3 col = wormEnv(d, u, tA, tB, w.fx.y, tMW, tRip);
    float rim = smoothstep(.82, 1.0, wu);
    col = mix(col, col * 1.35 + float3(.06, .08, .16), rim * W * .8);
    col *= mix(1.0, .2, smoothstep(.985, 1.0, wu));
    col *= mix(1.0, w.fx.z, w.wh.w);
    col += noise * w.fx.x * w.wh.w;
    return float4(clamp(dither(col, fc), 0.0, 1.0), 1.0);   // same dither as the garden (seamless cross)
}
